```java
var outputIterator = plan.output().iterator();
Attribute tsid = null, timestamp = null, score = null;

            case MetadataAttribute.TSID_FIELD:
                tsid = ma;
                break;
```

In `#GZRG9`, the logic for checking scoring attributes can be simplified. The use of a flag (`hasScoring`) and nested loops can be streamlined.
```java
if (filter != null) {
    boolean hasScoring = false;
    physicalPlan.forEachDown(EsQueryExec.class, esQueryExec -> {
        if (!hasScoring && esQueryExec.hasScoring()) {
            hasScoring = true;
        }
    });
    if (!hasScoring) {
        physicalPlan.forEachDown(FragmentExec.class, fragmentExec -> {
            fragmentExec.fragment().forEachDown(EsRelation.class, esRelation -> {
                if (!hasScoring && esRelation.output()
                    .stream()
                    .anyMatch(a -> a instanceof MetadataAttribute ma && ma.name().equals(MetadataAttribute.SCORE))) {
                    hasScoring = true;
                }
            });
        });
    }
    physicalPlan = physicalPlan.transformUp(FragmentExec.class, f -> {
        QueryBuilder fragmentFilter = f.esFilter();
        QueryBuilder newFilter;
        if (fragmentFilter != null) {
            newFilter = boolQuery().filter(fragmentFilter);
            if (hasScoring) {
                newFilter.must(filter);
            } else {
                newFilter.filter(filter);
            }
        } else {
            newFilter = hasScoring ? filter : boolQuery().filter(filter);
        }
        LOGGER.debug("Fold filter {} to EsQueryExec", newFilter);
        return f.withFilter(newFilter);
    });
}
```
